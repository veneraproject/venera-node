cmake_minimum_required(VERSION 3.16)
project(venera_local_cli LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl>=7.68)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
find_package(OpenSSL 1.1.1 REQUIRED)
find_package(Threads REQUIRED)
add_executable(venera-local-cli venera_cli_local.cpp native_evm.h local_wallet.h local_tls.h wallet_display.h wallet_session.h local_rewards.h json_compat.h)
target_link_libraries(venera-local-cli PRIVATE PkgConfig::CURL PkgConfig::JSONC OpenSSL::Crypto Threads::Threads)
 
 
target_compile_definitions(venera-local-cli PRIVATE OPENSSL_SUPPRESS_DEPRECATED)
install(TARGETS venera-local-cli RUNTIME DESTINATION bin)

if(WIN32)
  target_compile_definitions(venera-local-cli PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0602)
  target_link_libraries(venera-local-cli PRIVATE advapi32 ws2_32)
endif()

option(VENERA_NATIVE_TESTS "Build native cryptographic integration tests" OFF)
if(VENERA_NATIVE_TESTS)
  enable_testing()
  add_executable(venera-native-tests ../tests/native_crypto_tests.cpp ../daemon/src/crypto/keccak.c)
  target_include_directories(venera-native-tests PRIVATE ../daemon/contrib/epee/include)
  target_compile_definitions(venera-native-tests PRIVATE OPENSSL_SUPPRESS_DEPRECATED)
  target_link_libraries(venera-native-tests PRIVATE PkgConfig::CURL PkgConfig::JSONC OpenSSL::Crypto Threads::Threads)
  add_test(NAME native-crypto COMMAND venera-native-tests)
endif()
